"""Read-only post-stop memory and GGUF census. Never loads or executes a model."""
import collections
import datetime
import json
import pathlib
import struct
import subprocess

ROOT = pathlib.Path(__file__).resolve().parents[2]
MODEL = ROOT / 'archive/proof-s4-models/Muse-Glimmer-30B-KQuant-17GB-Q4_K_M.gguf'
OUT = ROOT / 'docs/plans/backfill-proof/evidence/s4-buyer-models/glimmer-cpu-authorized-2026-11-03-01'
SIZES = {0: 1, 1: 1, 2: 2, 3: 2, 4: 4, 5: 4, 6: 4, 7: 1, 10: 8, 11: 8, 12: 8}
FORMATS = {0: '<B', 1: '<b', 2: '<H', 3: '<h', 4: '<I', 5: '<i', 6: '<f', 7: '<?', 10: '<Q', 11: '<q', 12: '<d'}


def command(*args):
    return subprocess.check_output(args, text=True).strip()


with MODEL.open('rb') as f:
    def number(fmt):
        return struct.unpack(fmt, f.read(struct.calcsize(fmt)))[0]

    def string(keep=True):
        length = number('<Q')
        assert length <= MODEL.stat().st_size
        if keep and length < 16384:
            return f.read(length).decode('utf-8')
        f.seek(length, 1)
        return {'stringBytes': length}

    def value(kind, keep=True):
        if kind == 8:
            return string(keep)
        if kind == 9:
            element, length = number('<I'), number('<Q')
            if element in SIZES:
                f.seek(SIZES[element] * length, 1)
            else:
                for _ in range(length):
                    value(element, False)
            return {'arrayType': element, 'length': length}
        if keep:
            return number(FORMATS[kind])
        f.seek(SIZES[kind], 1)
        return None

    assert f.read(4) == b'GGUF'
    version, tensor_count, metadata_count = number('<I'), number('<Q'), number('<Q')
    assert version in (2, 3)
    metadata = {}
    for _ in range(metadata_count):
        key, kind = string(), number('<I')
        metadata[key] = value(kind)
    types = collections.Counter()
    tensor_elements = collections.Counter()
    for _ in range(tensor_count):
        string(False)
        dimensions = [number('<Q') for _ in range(number('<I'))]
        kind = number('<I')
        number('<Q')
        elements = 1
        for dim in dimensions:
            elements *= dim
        types[str(kind)] += 1
        tensor_elements[str(kind)] += elements
    alignment = metadata.get('general.alignment', 32)
    data_offset = (f.tell() + alignment - 1) // alignment * alignment

processes = []
for line in command('/bin/ps', '-axo', 'rss=,comm=').splitlines():
    rss, name = line.strip().split(None, 1)
    processes.append({'rssKiB': int(rss), 'executable': name})
processes.sort(key=lambda row: row['rssKiB'], reverse=True)
trials = []
for name in ['live-authorized-2026-11-03-01', 'glimmer-cpu-authorized-2026-11-03-01']:
    d = OUT.parent / name
    m = json.loads((d / 'memory-samples.json').read_text())
    samples = m['samples']
    trials.append({'directory': name, 'stopReason': m.get('memoryStopReason', m.get('stopReason')), 'samples': len(samples), 'peakSampledProcessRssKiB': max((int(row.get('rssKiB', row.get('rss', 0))) for row in samples), default=0), 'lastSample': samples[-1] if samples else None})
result = {
    'atUtc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'method': 'Static GGUF header/tensor inventory plus post-stop OS process census; no model load or inference',
    'premiseUnderReview': 'Changing GPU placement alone would make this same dense artifact load within the agreed memory-pressure boundary',
    'ramBytes': int(command('/usr/sbin/sysctl', '-n', 'hw.memsize')),
    'pressureLevelAfterStop': int(command('/usr/sbin/sysctl', '-n', 'kern.memorystatus_vm_pressure_level')),
    'swapAfterStop': command('/usr/sbin/sysctl', 'vm.swapusage'),
    'vmStatAfterStop': command('/usr/bin/vm_stat'),
    'largestResidentProcessesAfterStop': processes[:15],
    'trials': trials,
    'gguf': {'fileBytes': MODEL.stat().st_size, 'version': version, 'tensorCount': tensor_count, 'metadataCount': metadata_count, 'alignedTensorDataOffset': data_offset, 'tensorDataRegionBytes': MODEL.stat().st_size - data_offset, 'metadata': metadata, 'tensorTypeCounts': dict(types), 'tensorElementCountsByType': dict(tensor_elements)},
    'limits': ['Post-stop process census does not identify allocation owners at the failure peak.', 'RSS excludes some shared/GPU/compressed memory; it is not total RAM demand.', 'Tensor data bytes are disk bytes, not an established runtime working set.', 'No third load, altered guard, smaller model or runtime substitution is authorized.'],
}
(OUT / 'static-memory-census.json').write_text(json.dumps(result, indent=2) + '\n')
print(json.dumps({'architecture': metadata.get('general.architecture'), 'tensorCount': tensor_count, 'tensorDataRegionBytes': MODEL.stat().st_size - data_offset, 'pressureAfterStop': result['pressureLevelAfterStop'], 'additionalModelLoads': 0}))
