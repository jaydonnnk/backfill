import assert from 'node:assert/strict';
import { test } from 'node:test';
import { mkdtemp, writeFile, rm } from 'node:fs/promises';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { pathToFileURL } from 'node:url';
import { spawnSync } from 'node:child_process';
test('preload blocks direct Meta, allows exactly one pinned local request and refuses altered bodies before upstream', async () => {
  const dir = await mkdtemp(join(tmpdir(), 'backfill-s4-preload-test-'));
  try {
    const body = { model: 'muse-glimmer-s4', max_tokens: 512, temperature: 0, messages: [{ role: 'user', content: 'synthetic' }] };
    await writeFile(join(dir, 'current-case.json'), JSON.stringify({ name: 'eligible-1' }));
    await writeFile(join(dir, 'eligible-1-approved-glimmer.json'), JSON.stringify(body));
    const script = `
      import assert from 'node:assert/strict';
      let calls=0;
      globalThis.fetch=async()=>{calls++;return Response.json({fixture:true});};
      await import(${JSON.stringify(pathToFileURL(join(process.cwd(), 'scripts/verification/proof-s4-live-preload.mjs')).href)});
      await assert.rejects(fetch('https://api.meta.ai/v1/models'),/unavailable/);
      const init={method:'POST',redirect:'error',body:${JSON.stringify(JSON.stringify(body))}};
      await fetch('http://127.0.0.1:8088/v1/chat/completions',init);
      await assert.rejects(fetch('http://127.0.0.1:8088/v1/chat/completions',init),/EEXIST/);
      await assert.rejects(fetch('http://127.0.0.1:8088/v1/chat/completions',{...init,body:JSON.stringify({...JSON.parse(init.body),max_tokens:513})}));
      assert.equal(calls,1);
    `;
    const result = spawnSync(process.execPath, ['--input-type=module', '-e', script], { encoding: 'utf8', env: { PATH: process.env.PATH, DATABASE_URL: 'postgres://postgres@127.0.0.1:5432/postgres', S4_LIVE_RUN_DIR: dir, S4_LIVE_AUTHORITY: 'approved-5-calls-usd6.57' } });
    assert.equal(result.status, 0, result.stderr);
  } finally { await rm(dir, { recursive: true, force: true }); }
});
