import assert from 'node:assert/strict';
import { randomUUID } from 'node:crypto';
import { readFile, writeFile } from 'node:fs/promises';
import { join } from 'node:path';
import { chromium } from '@playwright/test';
import { eq } from 'drizzle-orm';
import type { Db } from '../../src/lib/db/client';
import { buyerSessions, commitments } from '../../src/lib/db/schema';
import { mintAgent } from '../../src/lib/agents/tokens';
import { getAgentTerms, settleDue } from '../../src/lib/commitments/service';
import { demoSnapshot } from '../../src/lib/demo/snapshot';
import { Intent, gatewayBuyer, glimmerParser } from '../../src/lib/buyer/models';
import { Session } from '../../src/lib/buyer/service';
import { buyerRecording, replayBuyer } from '../../src/lib/buyer/recording';
import { z } from 'zod';
const Plan = z.object({ cases: z.array(z.object({ id: z.string(), request: z.string(), partySize: z.number(), budgetPerCoverCents: z.number().nullable() })) });
export async function proveS4Buyers({ db, appUrl, directory, mainRuleId, refusalRuleId, ownerId, glimmerAvailable }: { db: Db; appUrl: string; directory: string; mainRuleId: string; refusalRuleId: string; ownerId: string; glimmerAvailable: () => boolean }) {
  const plan = Plan.parse(JSON.parse(await readFile(join(directory, '../proposed-inputs.json'), 'utf8')));
  const browser = await chromium.launch({ headless: true });
  const page = await browser.newPage({ viewport: { width: 1280, height: 900 } });
  const results: { case: string; session: Session; parserMatched: boolean | null }[] = [];
  page.setDefaultTimeout(70_000);
  const errors: string[] = []; page.on('pageerror', e => errors.push(e.name));
  async function stored(partyId: string) {
    const [row] = await db.select().from(buyerSessions).where(eq(buyerSessions.partyId, partyId));
    return Session.parse(row?.record);
  }
  async function click(name: string) {
    const response = page.waitForResponse(r => r.url() === `${appUrl}/api/buyer` && r.request().method() === 'POST');
    await page.getByRole('button', { name, exact: true }).click();
    const r = await response;
    assert.equal(r.status(), 200);
    return r.json();
  }
  async function capture(name: string) {
    for (const width of [1280, 375]) {
      await page.setViewportSize({ width, height: 900 });
      assert(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth));
      const png = await page.screenshot({ path: join(directory, `${name}-${width}.png`), fullPage: true });
      assert.equal(png.readUInt32BE(16), width);
    }
  }
  // Preparation uses a dummy credential and an intercepting transport, never the real key/network.
  const captureRequest = async (work: (fetcher: typeof fetch) => Promise<unknown>) => {
    let body: unknown;
    try { await work(async (_url, init) => { body = JSON.parse(String(init?.body)); throw Error('offline_request_preparation'); }); } catch { /* Expected offline interception. */ }
    assert(body, 'SDK request preparation failed before transport');
    return body;
  };
  try {
    for (const c of plan.cases) {
      const ambiguous = c.id === 'ambiguous-local-only';
      if (ambiguous && !glimmerAvailable()) { await writeFile(join(directory, `${c.id}-blocked.json`), JSON.stringify({ status: 'glimmer_unavailable', gatewayAttempts: 0 })); continue; }
      const party = await mintAgent(db, randomUUID(), `Synthetic S4 genuine test ${c.id}`);
      await writeFile(join(directory, 'current-case.json'), JSON.stringify({ name: c.id }));
      await page.goto(`${appUrl}/buyer`);
      await page.getByLabel('Buyer credential', { exact: true }).fill(party.token);
      await page.getByLabel('Your request', { exact: true }).fill(c.request);
      await click('Save request');
      let s = await stored(party.id);
      let parserMatched: boolean | null = null;
      if (glimmerAvailable()) {
        const body = await captureRequest(fetcher => glimmerParser('http://127.0.0.1:8088/v1', 'muse-glimmer-s4', fetcher)(s.request, s.referenceClock));
        await writeFile(join(directory, `${c.id}-approved-glimmer.json`), JSON.stringify(body, null, 2), { flag: 'wx' });
        await click('Ask local Glimmer'); s = await stored(party.id);
        await writeFile(join(directory, `${c.id}-parsed.json`), JSON.stringify(s, null, 2), { flag: 'wx' });
        if (ambiguous) {
          parserMatched = s.phase === 'clarification' && s.proposal?.partySize === 2 && s.proposal.budgetPerCoverCents === null && s.proposal.earliest === null && s.proposal.latest === null;
          await capture('genuine-clarification'); results.push({ case: c.id, session: s, parserMatched }); continue;
        }
        parserMatched = s.proposal?.partySize === 2 && s.proposal.visitDate === '2026-11-05' && s.proposal.earliest === '14:00' && s.proposal.latest === '17:00' && s.proposal.currency === 'SGD' && s.proposal.budgetPerCoverCents === c.budgetPerCoverCents;
      }
      const intent = Intent.parse({ partySize: 2, visitDate: '2026-11-05', earliest: '14:00', latest: '17:00', budgetPerCoverCents: c.budgetPerCoverCents, currency: 'SGD', requirements: s.proposal?.requirements ?? [] });
      await page.getByLabel('Party size', { exact: true }).fill('2');
      await page.getByLabel('Singapore visit date', { exact: true }).fill(intent.visitDate!);
      await page.getByLabel('Earliest arrival', { exact: true }).fill(intent.earliest!);
      await page.getByLabel('Latest departure', { exact: true }).fill(intent.latest!);
      await page.getByLabel('Maximum per cover, in cents', { exact: true }).fill(String(intent.budgetPerCoverCents));
      await page.getByLabel('Budget currency', { exact: true }).selectOption('SGD');
      await click('Save reviewed constraints'); s = await stored(party.id);
      assert.equal(s.phase, 'reviewed'); assert.deepEqual(s.constraints, intent);
      const ruleId = c.id === 'refusal-1' ? refusalRuleId : mainRuleId;
      const terms = await getAgentTerms(db, ruleId); assert(terms);
      const offer = { merchantName: terms.merchantName, about: 'the public offer summary; no additional dietary or location guarantees', summary: terms.summary, ...terms.nextOccurrence, perAgentCapCovers: terms.offer.perAgentCapCovers };
      const body = z.object({ prompt: z.unknown() }).parse(await captureRequest(fetcher => gatewayBuyer(fetcher)(party.id, s.request, intent, offer)));
      await writeFile(join(directory, `${c.id}-approved-request.json`), JSON.stringify({ prompt: body.prompt, partyId: party.id, sessionId: s.id, reviewedConstraints: intent, publicOffer: offer }, null, 2), { flag: 'wx' });
      const offersResponse = page.waitForResponse(r => r.url() === `${appUrl}/api/buyer` && r.request().method() === 'GET');
      await page.getByRole('button', { name: 'Load public offers', exact: true }).click(); await offersResponse;
      await page.getByLabel('Offer', { exact: true }).selectOption(ruleId);
      await click('Ask Spark Open for a decision'); s = await stored(party.id);
      await writeFile(join(directory, `${c.id}-decision.json`), JSON.stringify(s, null, 2), { flag: 'wx' });
      const httpPath = join(directory, `${c.id}-http.json`);
      try { const http = JSON.parse(await readFile(httpPath, 'utf8')); assert(![402, 403].includes(http.status), 'Account/billing gate: no more dispatches'); } catch (e) { if (e instanceof assert.AssertionError) throw e; }
      assert.equal((await db.select().from(commitments).where(eq(commitments.agentId, party.id))).length, 0);
      if (s.phase !== 'decided' || !s.decision) { results.push({ case: c.id, session: s, parserMatched }); continue; }
      if (c.id === 'refusal-1') {
        assert.equal(s.decision.covers, 0); assert.equal(s.approval, null);
        await capture('genuine-refusal'); results.push({ case: c.id, session: s, parserMatched }); continue;
      }
      assert.equal(s.decision.covers, 2);
      if (c.id === 'eligible-1') await capture('genuine-unapproved');
      await click('Approve these exact terms'); s = await stored(party.id); assert(s.approval); assert.equal(s.result, null);
      await click('Submit approved commitment'); s = await stored(party.id); assert(s.result?.ok);
      const duplicate = await page.request.post(`${appUrl}/api/buyer`, { headers: { Authorization: `Bearer ${party.token}` }, data: { action: 'submit', id: s.id, revision: s.revision } });
      assert.deepEqual((await duplicate.json()).session.result, s.result);
      assert.equal((await db.select().from(commitments).where(eq(commitments.agentId, party.id))).length, 1);
      const recording = buyerRecording(s); const replay = replayBuyer(recording);
      assert.deepEqual((await replay.decide(party.id, recording.request, recording.constraints, recording.offer)).value, recording.decision);
      await assert.rejects(replay.decide('wrong-party', recording.request, recording.constraints, recording.offer));
      await assert.rejects(replay.decide(party.id, recording.request, recording.constraints, { ...recording.offer, visitDate: '2026-11-12' }));
      if (recording.parsed) assert.deepEqual((await replay.parse(recording.request, recording.referenceClock)).value, recording.parsed);
      else await assert.rejects(replay.parse(recording.request, recording.referenceClock), /missing_parser_recording/);
      await writeFile(join(directory, `${c.id}-final.json`), JSON.stringify({ session: s, recording, offlineReplayVerified: true, approvals: 'scripted authenticated UI actions, not real-person consent' }, null, 2), { flag: 'wx' });
      results.push({ case: c.id, session: s, parserMatched });
      if (c.id === 'eligible-1') await capture('genuine-pending');
    }
    const pending = await demoSnapshot(db); const eligible = results.filter(r => r.case.startsWith('eligible-') && r.session.result?.ok);
    assert.equal(pending.scene?.pooled.filledCovers, 0); assert.equal(pending.scene?.pooled.covers, eligible.length * 2);
    await settleDue(db, ownerId, () => new Date('2026-11-05T12:00:01+08:00'));
    const settled = await demoSnapshot(db, new Date('2026-11-05T12:00:01+08:00'));
    assert.equal(settled.scene?.pooled.filledCovers, eligible.length === 4 ? 8 : 0);
    await page.goto(`${appUrl}/demo`); await capture('genuine-settled');
    assert.deepEqual(errors, []);
    const audit = { scope: 'genuine model output, synthetic parties and menu, scripted actual UI review/approval, actual MCP and disposable PostgreSQL; deadline advanced through injected service clock', results, pending, settled, errors, s4aPassed: results.length === 6 && results.every(r => r.parserMatched === true), s4bPassed: eligible.length === 4 && results.some(r => r.case === 'refusal-1' && r.session.decision?.covers === 0) };
    await writeFile(join(directory, 'ui-and-database-audit.json'), JSON.stringify(audit, null, 2), { flag: 'wx' });
    return audit;
  } finally { await browser.close(); }
}
